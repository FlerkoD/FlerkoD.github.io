🔧 SQL-скрипт создания и заполнения


        -- Создание базы данных (если не существует)
CREATE DATABASE IF NOT EXISTS test_finance;
USE test_finance;

-- -----------------------------------------------------
-- Создание таблиц
-- -----------------------------------------------------

CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    login VARCHAR(50) NOT NULL,
    email VARCHAR(100),
    full_name VARCHAR(100),
    birth_date DATE,
    registration_date DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE products (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    category VARCHAR(50),
    price DECIMAL(10,2) NOT NULL
);

CREATE TABLE orders (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NULL,
    order_date DATETIME,
    total DECIMAL(10,2),
    status VARCHAR(20),
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE SET NULL
);

CREATE TABLE order_items (
    id INT PRIMARY KEY AUTO_INCREMENT,
    order_id INT NOT NULL,
    product_id INT NOT NULL,
    quantity INT NOT NULL,
    price DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (order_id) REFERENCES orders(id) ON DELETE CASCADE,
    FOREIGN KEY (product_id) REFERENCES products(id)
);

CREATE TABLE employees (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100),
    department VARCHAR(50),
    salary DECIMAL(10,2)
);

CREATE TABLE transactions (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    amount DECIMAL(10,2),
    transaction_date DATETIME,
    type VARCHAR(20),
    FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE TABLE user_events (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    event_time DATETIME,
    event_type VARCHAR(50),
    FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE TABLE user_interests (
    user_id INT NOT NULL,
    category VARCHAR(50),
    FOREIGN KEY (user_id) REFERENCES users(id)
);

-- Генерация пользователей (100 записей)
INSERT INTO users (login, email, full_name, birth_date, registration_date)
WITH RECURSIVE nums AS (SELECT 1 AS n UNION ALL SELECT n+1 FROM nums WHERE n < 100)
SELECT
    CONCAT('user', FLOOR(RAND() * 30) + 1) AS login,  -- дубли логинов (около 30 уникальных)
    CONCAT('user', FLOOR(RAND() * 100) + 1, '@example.com') AS email, -- могут быть дубли
    CONCAT('Full Name ', FLOOR(RAND() * 1000)) AS full_name,
    DATE_SUB(CURDATE(), INTERVAL (18 + FLOOR(RAND() * 50)) YEAR) AS birth_date,
    DATE_SUB(NOW(), INTERVAL FLOOR(RAND() * 730) DAY) AS registration_date
FROM nums;

-- Генерация заказов (100 записей)
INSERT INTO orders (user_id, order_date, total, status)
WITH RECURSIVE nums AS (SELECT 1 AS n UNION ALL SELECT n+1 FROM nums WHERE n < 100)
SELECT
    CASE WHEN RAND() < 0.1 THEN NULL ELSE FLOOR(RAND() * 100) + 1 END AS user_id, -- ~10% висяков
    DATE_SUB(NOW(), INTERVAL FLOOR(RAND() * 365) DAY) AS order_date,
    ROUND(RAND() * 1000, 2) * IF(RAND() < 0.05, -1, 1) AS total, -- 5% отрицательных
    ELT(FLOOR(RAND() * 3) + 1, 'completed', 'pending', 'cancelled') AS status
FROM nums;

-- Генерация позиций заказов (100 записей)

    FLOOR(RAND() * 100) + 1 AS order_id,
    FLOOR(RAND() * 20) + 1 AS product_id,
    FLOOR(RAND() * 5) + 1 AS quantity,
    ROUND(RAND() * 500 + 5, 2) AS price -- произвольная цена, иногда не совпадает с product.price
FROM nums;

-- Генерация транзакций (100 записей, с отрицательными суммами для withdrawals)
INSERT INTO transactions (user_id, amount, transaction_date, type)
WITH RECURSIVE nums AS (SELECT 1 AS n UNION ALL SELECT n+1 FROM nums WHERE n < 100)
SELECT
    FLOOR(RAND() * 100) + 1 AS user_id,
    ROUND(RAND() * 1000 - 200, 2) AS amount, -- диапазон -200..800
    DATE_SUB(NOW(), INTERVAL FLOOR(RAND() * 90) DAY) AS transaction_date,
    IF(RAND() < 0.3, 'withdrawal', 'payment') AS type
FROM nums;

-- Генерация событий пользователей (100 записей, для сессий и неактивности)
INSERT INTO user_events (user_id, event_time, event_type)
WITH RECURSIVE nums AS (SELECT 1 AS n UNION ALL SELECT n+1 FROM nums WHERE n < 100)
SELECT
    FLOOR(RAND() * 100) + 1 AS user_id,
    DATE_SUB(NOW(), INTERVAL FLOOR(RAND() * 14) DAY) + INTERVAL FLOOR(RAND() * 24) HOUR + INTERVAL FLOOR(RAND() * 60) MINUTE AS event_time,
    ELT(FLOOR(RAND() * 5) + 1, 'login', 'logout', 'view', 'add_to_cart', 'purchase') AS event_type
FROM nums;

-- Генерация интересов пользователей (для 50 пользователей, по 2-3 категории)
INSERT INTO user_interests (user_id, category)
WITH RECURSIVE nums AS (SELECT 1 AS n UNION ALL SELECT n+1 FROM nums WHERE n < 150) -- 150 записей
SELECT
    FLOOR(RAND() * 50) + 1 AS user_id, -- только первые 50 пользователей
    ELT(FLOOR(RAND() * 5) + 1, 'Electronics', 'Books', 'Clothing', 'Sports', 'Home') AS category
FROM nums
LIMIT 150; -- у некоторых пользователей будет по 2-3 интереса
