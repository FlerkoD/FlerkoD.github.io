-- =====================================================
-- ЛЕТНЯЯ БАЗА ДАННЫХ ДЛЯ SQL-АДВЕНТА (SummerDB)
-- =====================================================
DROP DATABASE IF EXISTS SummerDB;
CREATE DATABASE SummerDB CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE SummerDB;

CREATE TABLE tourists (
  tourist_id INT AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(100) NOT NULL,
  city VARCHAR(100),
  age INT
);

CREATE TABLE souvenirs (
  souvenir_id INT AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(150) NOT NULL,
  category VARCHAR(50),
  price DECIMAL(8,2),
  stock INT
);

CREATE TABLE bookings (
  booking_id INT AUTO_INCREMENT PRIMARY KEY,
  tourist_id INT NOT NULL,
  souvenir_id INT NOT NULL,
  quantity INT NOT NULL,
  booking_date DATE,
  processed BOOLEAN DEFAULT FALSE,
  gift_wrapped BOOLEAN DEFAULT FALSE,
  FOREIGN KEY (tourist_id) REFERENCES tourists(tourist_id),
  FOREIGN KEY (souvenir_id) REFERENCES souvenirs(souvenir_id)
);

CREATE TABLE beach_items (
  item_id INT AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(100),
  color VARCHAR(50),
  material VARCHAR(50),
  price DECIMAL(6,2)
);

CREATE TABLE summer_songs (
  song_id INT AUTO_INCREMENT PRIMARY KEY,
  title VARCHAR(150),
  artist VARCHAR(100),
  mood VARCHAR(50),
  length_seconds INT
);

CREATE TABLE summer_movies (
  movie_id INT AUTO_INCREMENT PRIMARY KEY,
  title VARCHAR(200),
  year INT,
  genre VARCHAR(50),
  rating DECIMAL(3,1)
);

CREATE TABLE trips (
  trip_id INT AUTO_INCREMENT PRIMARY KEY,
  tourist_id INT,
  trip_date DATE,
  description TEXT,
  FOREIGN KEY (tourist_id) REFERENCES tourists(tourist_id)
);
